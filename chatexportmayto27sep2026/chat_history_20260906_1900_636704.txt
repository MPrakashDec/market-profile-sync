def resolve_contracts():

    """

    Dynamically resolves the active near-month Nifty Futures contract,

    the active nearest weekly options series, and maps all available strikes.

    """

    now_ts = time.time()

    nifty_options = []

    future_futs = []



    if not os.path.exists(SECURITY_MASTER_CSV):

        print(f"[RESOLVER ERROR] Security master CSV missing at {SECURITY_MASTER_CSV}", flush=True)

        return {}, 0, "N/A", "NSE:NIFTY26SEPFUT"



    try:

        with open(SECURITY_MASTER_CSV, "r", encoding="utf-8", errors="ignore") as f:

            for line in f:

                cols = line.strip().split(",")

                if len(cols) > 16 and cols[13] == "NIFTY":

                    if "FUT" in cols[9] and len(cols) > 8 and cols[8]:

                        try:

                            exp = int(cols[8])

                            if exp >= now_ts:

                                future_futs.append({"ticker": cols[9], "expiry_ts": exp})

                        except ValueError:

                            pass

                    elif cols[16] in ["CE", "PE"] and len(cols) > 8 and cols[8]:

                        try:

                            exp = int(cols[8])

                            if exp >= now_ts:

                                nifty_options.append({

                                    "ticker": cols[9],

                                    "strike": float(cols[15]),

                                    "type": cols[16],

                                    "expiry_ts": exp

                                })

                        except ValueError:

                            pass



        future_futs.sort(key=lambda x: x["expiry_ts"])

        near_fut = future_futs[0]["ticker"] if future_futs else "NSE:NIFTY26SEPFUT"



        unique_expiries = sorted(list(set([o["expiry_ts"] for o in nifty_options])))

        if not unique_expiries:

            return {}, 0, "N/A", near_fut



        nearest_expiry = unique_expiries[0]

        nearest_expiry_dt = datetime.datetime.fromtimestamp(nearest_expiry)

        expiry_label = nearest_expiry_dt.strftime("%d-%b").upper()



        nearest_opts = [o for o in nifty_options if o["expiry_ts"] == nearest_expiry]

        ticker_map = {}

        for o in nearest_opts:

            ticker_map[(int(o["strike"]), o["type"])] = o["ticker"]



        return ticker_map, nearest_expiry, expiry_label, near_fut

    except Exception as e:

        print(f"[RESOLVER EXCEPTION] {e}", flush=True)

        return {}, 0, "N/A", "NSE:NIFTY26SEPFUT"

def calculate_greeks_and_zscore(spot, fut, qmap, option_symbols):

    """

    Computes real-time dynamic Volume PCR, GEX, and analytical positioning.

    """

    ce_vols = []

    pe_vols = []



    for sym in option_symbols:

        vol = qmap.get(sym, {}).get("volume", 0)

        if sym.endswith("CE"):

            ce_vols.append(vol)

        elif sym.endswith("PE"):

            pe_vols.append(vol)



    tot_ce = sum(ce_vols) or 1

    tot_pe = sum(pe_vols) or 1

    pcr_vol = tot_pe / tot_ce



    gex_score = (tot_pe - tot_ce) / 1000000.0

    vanna_bias = "Positive (Dealers Long OTM Put Vanna)" if fut >= spot else "Negative (Put Buying Overhang)"

    charm_bias = "Bullish Drift (Accelerating Put Theta Decay)" if fut >= spot else "Neutral/Defensive"

    basis = fut - spot

    basis_z = (basis - 60.0) / 20.0

    quant_z = np.clip((basis_z * 0.4) + ((pcr_vol - 1.0) * 1.5) + (1.0 if fut >= spot else -1.0), -3.0, 3.0)

    return pcr_vol, gex_score, vanna_bias, charm_bias, basis, quant_z



def pull_tpo_periods(fyers, fut_ticker):

    today = datetime.datetime.now().strftime("%Y-%m-%d")

    period_data = {}

    for name, start_t, end_t in PERIOD_BOUNDS:

        period_data[name] = {"high": -1, "low": 999999, "open": None, "close": None, "candles": 0}



    try:

        res_fut = fyers.history(data={

            "symbol": fut_ticker,

            "resolution": "5",

            "date_format": "1",

            "range_from": today,

            "range_to": today,

            "cont_flag": "1"

        })

        candles = res_fut.get("candles", [])



        for c in candles:

            dt = datetime.datetime.fromtimestamp(c[0])

            t = dt.time()

            for name, start_t, end_t in PERIOD_BOUNDS:

                if start_t <= t < end_t:

                    p = period_data[name]

                    if p["open"] is None:

                        p["open"] = c[1]

                    p["high"] = max(p["high"], c[2])

                    p["low"] = min(p["low"], c[3])

                    p["close"] = c[4]

                    p["candles"] += 1

                    break

    except Exception as e:

        print(f"[TPO HISTORY ERROR] {e}", flush=True)



    return period_data
